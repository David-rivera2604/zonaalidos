CREATE TABLE WEBHOOK 
   (	ID NUMBER(38,0), 
	CALLBACK VARCHAR2(512), 
	TYPE NUMBER(9,0), 
	RESTENDPOINTID NUMBER(9,0), 
	RETRYCOUNT NUMBER(5,0) DEFAULT 3, 
	MAPPER VARCHAR2(2024), 
	ACTIVE NUMBER(1,0) DEFAULT 1, 
	TEMPLATE CLOB, 
	 CONSTRAINT WEBHOOK_PK PRIMARY KEY (ID)
  USING INDEX  ENABLE, 
	 CONSTRAINT FK_WEBHOOKKIND FOREIGN KEY (ID)
	  REFERENCES WEBHOOKKIND (ID) ENABLE, 
	 CONSTRAINT FK_RESTENDPOINTS FOREIGN KEY (ID)
	  REFERENCES RESTENDPOINTS (ID) ENABLE
   ) ;

   (	ID NUMBER(5,0) NOT NULL ENABLE,

	REQUEST CLOB, 
	UPDATEDATE DATE, 
	TYPEID NUMBER(5,0), 
	 CONSTRAINT WEBHOOKTRACK_PK PRIMARY KEY (ID)
  USING INDEX  ENABLE
   ) ;
