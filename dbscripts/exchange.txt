SELECT CS.UPDATEDATE, C.DESCRIPTION TENANT, CS.ENTITYID REF, CS."ACTION" || '. ' || CS.SUMMARY SUMMARY, a1332.cod_agt, u.FIRSTNAME || ' ' || u.LASTNAME AS USERNAME , u.EMAIL
FROM ALIADOS.CHANGESET cs  
    LEFT JOIN ALIADOS.LOOKUP C ON C.LOOKUPID=4 AND C.LANGUAGE=1 AND C.CODE = cs.COMPANYID
    LEFT JOIN ALIADOS.USERMEMBER u ON u.USERID = cs.UPDATEUSERCODE 
    LEFT JOIN a1001332 a1332 ON a1332.cod_cia = 1 AND a1332.email_com = u.EMAIL 
WHERE cs.ENTITYTYPE = 3000 
ORDER BY cs.id desc;
